Google Sheets Setup

Follow these steps once and your product catalog and slideshow will automatically pull from your Google Sheet. Just edit the sheet — the website updates itself.

  1. 1
    Your Sheet is already connected

    Your Google Sheet CSV URL is hardcoded in this file. To change it, open index.html and update SHEET_CSV_URL at the top of the script.

  2. 2
    Sheet column headers
    ColumnRequired?What to put in it
    nameRequiredProduct name
    descriptionRequiredFull product description
    animalOptionalCattle, Sheep, Goats, Swine, Horses…
    styleOptionalButton, Flag, Two-Piece, Visual ID…
    sizeOptionalSmall, Medium, Large, Extra Large…
    colorOptionalYellow, Green, Red, Blue, Orange…
    priceOptionale.g. $1.25 — shows as "$1.25/per Tag with Engraving"
    badgeOptionalPopular, New, Sale
    inventoryOptionalNumber, "In Stock", "Low Stock", or "Out of Stock"
    image_urlOptionalDirect URL to a product photo
    slideshow_urlOptionalPhoto or .mp4 URL for the hero slideshow
    slideshow_typeOptionalLeave blank for photos; put video for .mp4
    slide_titleOptionalHeadline text on the slide
    slide_subtitleOptionalSmaller caption under the headline

  4. 4
    Connect Google Apps Script (cross-device hit counter & login log)

    ① In your Google Sheet → Extensions → Apps Script → delete existing code and paste:

    function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const action = e.parameter.action || ''; if (action === 'hit') { let sh = ss.getSheetByName('Hits') || ss.insertSheet('Hits'); if (sh.getLastRow() === 0) sh.appendRow(['Count']); if (sh.getLastRow() < 2) sh.appendRow([0]); const c = sh.getRange('A2'); c.setValue((c.getValue()||0)+1); return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON); } if (action === 'getHits') { const sh = ss.getSheetByName('Hits'); const count = sh ? (sh.getRange('A2').getValue()||0) : 0; return ContentService.createTextOutput(JSON.stringify({count})).setMimeType(ContentService.MimeType.JSON); } if (action === 'logLogin') { let sh = ss.getSheetByName('LoginLog') || ss.insertSheet('LoginLog'); if (sh.getLastRow() === 0) sh.appendRow(['Time','Type','Screen','Browser','OS','City','Country','IP']); const d = JSON.parse(e.parameter.data||'{}'); sh.insertRowAfter(1); sh.getRange(2,1,1,8).setValues([[d.time||'',d.type||'',d.screen||'',d.browser||'',d.os||'',d.city||'',d.country||'',d.ip||'']]); return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON); } if (action === 'getLogs') { const sh = ss.getSheetByName('LoginLog'); if (!sh) return ContentService.createTextOutput('[]').setMimeType(ContentService.MimeType.JSON); const rows = sh.getDataRange().getValues().slice(1).map(r=>({time:r[0],type:r[1],screen:r[2],browser:r[3],os:r[4],city:r[5],country:r[6],ip:r[7]})); return ContentService.createTextOutput(JSON.stringify(rows)).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput('{}').setMimeType(ContentService.MimeType.JSON); }
